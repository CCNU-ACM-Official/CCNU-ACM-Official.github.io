<script setup>
    import {
        VPTeamPage,
        VPTeamPageTitle,
        VPTeamMembers,
        VPTeamPageSection
    } from 'vitepress/theme';
    import { groupByYear } from './.vitepress/theme/member-utils.js';

    // 一名成员一个数据文件，文件名就是 id（年份 + 当年顺序号）。
    // 页面不再手写数组：新增/修改只改 data 里的 json，这里永远不用动。
    const memberModules = import.meta.glob('./_data/members/*.json', { eager: true });

    const allMembers = Object.values(memberModules)
        .map((mod) => mod.default)
        .filter((member) => member && member.hidden !== true);

    // 只渲染真的有成员的年级；没有成员的年份不显示空标题（与迁移前版式一致）。
    const sections = groupByYear(allMembers);
</script>

<VPTeamPage>
    <VPTeamPageTitle>
        <template #title>CCNU ACM协会</template>
        <template #lead>太庙</template>
    </VPTeamPageTitle>
    <VPTeamPageSection v-for="section in sections" :key="section.year">
        <template #title>{{ section.year }}级</template>
        <template #lead>...</template>
        <template #members>
            <VPTeamMembers size="small" :members="section.members"/>
        </template>
    </VPTeamPageSection>
</VPTeamPage>
